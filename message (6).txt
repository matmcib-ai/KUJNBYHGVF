-- ============================================================
--  Dragging
-- ============================================================

local dragging, dragStart, startPos

local function onInputBegan(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        dragging  = true
        dragStart = input.Position
        startPos  = frame.Position
    end
end

local function onInputChanged(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
    or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        frame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end

local function onInputEnded(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end

frame.InputBegan:Connect(onInputBegan)
frame.InputChanged:Connect(onInputChanged)
frame.InputEnded:Connect(onInputEnded)
UserInputService.InputChanged:Connect(onInputChanged)

local currentFps = 0
local fpsBuffer = {}
RunService.RenderStepped:Connect(function(dt)
    table.insert(fpsBuffer, 1 / dt)
    if #fpsBuffer > 10 then table.remove(fpsBuffer, 1) end
    local sum = 0
    for _, v in ipairs(fpsBuffer) do sum += v end
    currentFps = math.round(sum / #fpsBuffer)
end)

local function getPing()
    local ok, val = pcall(function()
        return math.round(Players.LocalPlayer:GetNetworkPing() * 1000)
    end)
    if ok and val then return val end
    local ok2, val2 = pcall(function()
        return math.round(game:GetService("Stats").Network.ServerStatsItem["Data Ping"].Value)
    end)
    return (ok2 and val2) or 0
end

task.spawn(function()
    while true do
        local ping = getPing()
        local fpsColor = currentFps >= 50 and Color3.fromRGB(80, 220, 100)
            or currentFps >= 30 and Color3.fromRGB(255, 200, 50)
            or Color3.fromRGB(255, 80, 80)
        statsLabel.Text = "FPS: " .. currentFps .. "  |  Ping: " .. ping .. " ms"
        statsLabel.TextColor3 = fpsColor
        task.wait(0.5)
    end
end)

-- Attach a Focused listener to a TextBox
local function watchTextBox(obj)
    if not obj:IsA("TextBox") then return end
    if obj:IsDescendantOf(screenGui) then return end
    obj.Focused:Connect(function()
        if not pickMode then return end
        targetBox = obj
        stopPicking()
        setStatus("Target: " .. obj.Name, Color3.fromRGB(80, 255, 120))
    end)
end

-- Watch all current and future TextBoxes
for _, obj in ipairs(PlayerGui:GetDescendants()) do
    watchTextBox(obj)
end
PlayerGui.DescendantAdded:Connect(function(obj)
    task.wait() -- let it fully initialize
    watchTextBox(obj)
end)

-- ============================================================
--  Paste code into selected box
-- ============================================================

local function pasteCode(text)
    local code = text:gsub("%s+", "")
    if code == "" then return end

    if not targetBox then
        setStatus("No box selected â click Pick Box", Color3.fromRGB(255, 80, 80))
        return
    end

    if not targetBox.Parent then
        setStatus("Box gone â pick again", Color3.fromRGB(255, 80, 80))
        targetBox = nil
        return
    end

    targetBox.Text = code
    setStatus("Pasted: " .. code, Color3.fromRGB(80, 255, 120))
    task.delay(5, function()
        if statusLabel.Text == "Pasted: " .. code then
            setStatus("Target: " .. (targetBox and targetBox.Name or "none"), Color3.fromRGB(80, 255, 120))
        end
    end)
end

local REMOTE_HASH = "080efa832657457aaa9b706416b0366890af7efafc5243f40f2833036262119e"
local hooked = false

local function hookRemote(remote)
    if hooked then return end
    hooked = true
    print("[AutoRedeem] Hooked: " .. remote:GetFullName())
    setStatus("Listening â pick a box!", Color3.fromRGB(80, 255, 120))
    remote.OnClientEvent:Connect(function(msg)
        print("[AutoRedeem] Fired! msg=" .. tostring(msg))
        if type(msg) ~= "string" then return end
        local code = msg:gsub("%s+", "")
        if #code < 1 then return end
        pasteCode(msg)
    end)
end

-- Try scanning everything already in RS right now
local function scanNow()
    local function scan(parent)
        for _, child in ipairs(parent:GetChildren()) do
            if child:IsA("RemoteEvent") and child.Name:find(REMOTE_HASH, 1, true) then
                hookRemote(child)
                return
            end
            pcall(scan, child)
        end
    end
    scan(RS)
end

-- Also watch for it appearing later (replication delay)
RS.DescendantAdded:Connect(function(obj)
    if hooked then return end
    if obj:IsA("RemoteEvent") and obj.Name:find(REMOTE_HASH, 1, true) then
        hookRemote(obj)
    end
end)

-- Run immediately, then retry a few times with delays
task.spawn(function()
    for i = 1, 10 do
        scanNow()
        if hooked then break end
        task.wait(1)
    end
    if not hooked then
        warn("[AutoRedeem] Could not find global message remote after 10s.")
        setStatus("Remote not found â rejoin?", Color3.fromRGB(255, 80, 80))
    end
end)
